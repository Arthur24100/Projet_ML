# requirements.txt — conteneur api
# ==================================
# Dépendances nécessaires pour faire tourner l'API FastAPI.
# Plus complet que le requirements.txt du model-builder car l'API fait
# du preprocessing (pandas, numpy) en plus du chargement du modèle.

# fastapi : framework web asynchrone pour construire l'API REST.
# Gère le routing (/predict, /options, /health), la validation des requêtes
# via Pydantic, et génère automatiquement la doc Swagger à /docs.
fastapi==0.104.1

# uvicorn : serveur ASGI qui fait tourner FastAPI.
# ASGI (vs WSGI) supporte nativement l'asynchrone — important pour gérer
# plusieurs requêtes simultanées sans bloquer.
# Lancé via : uvicorn main:app --host 0.0.0.0 --port 8000
uvicorn==0.24.0

# pandas : utilisé dans preprocess() pour parser les dates (pd.to_datetime),
# calculer les features temporelles (.dt.day, .dt.dayofweek, etc.) et construire
# le DataFrame d'entrée du modèle. Également utilisé pour les constantes INDIAN_HOLIDAYS.
pandas==2.2.2

# numpy : utilisé pour expm1() (transformation inverse du log sur la prédiction),
# np.isnan() (détection des NaN dans le vecteur de features) et les calculs
# intermédiaires de preprocessing.
# IMPORTANT : doit correspondre à la version utilisée sur ta machine pour créer
# les .pkl — une différence de version numpy cause l'erreur "PCG64 is not a known BitGenerator".
numpy==2.0.2

# scikit-learn : nécessaire pour désérialiser les objets sklearn contenus dans les .pkl
# (StackingRegressor, StandardScaler, LabelEncoder) et appeler .predict() / .transform().
# Même contrainte de version que numpy — doit correspondre à la machine d'entraînement.
scikit-learn==1.5.2

# joblib : utilisé pour charger les .pkl au démarrage (joblib.load).
# C'est joblib qui a sérialisé les objets dans train_and_save.py,
# donc c'est joblib qui doit les désérialiser ici.
joblib==1.3.2
